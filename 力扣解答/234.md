我们采用获得中间点 反转后半部分和前半部分对比 然后再把后半部分再反转恢复 来达到空间复杂度O(1)

```
var isPalindrome = function(head) {

    //快慢双指针+反转链表

    let fast=head,slow=head

    while(fast&&fast.next){

        fast=fast.next.next

        slow=slow.next

    }

    let cur=right=reverse(slow),left=head//这样链表不会断链的 slow左侧的next指针一直指向slow  反转不反转都没有变化的

    while(cur){//如果为奇数 右半部分会多一个元素 我们比较左侧是否均满足即可 最后没有比较的是中间元素 不需要比较

        if(cur.val!==left.val)

            return false

        cur=cur.next

        left=left.next

    }

    reverse(right)//再恢复 反转回去

    return true

  
  

};
//反转右半部分
    function reverse(head){

        let pre=null,cur=head

        while(cur){

            let next=cur.next

            cur.next=pre

            pre=cur

            cur=next

        }

        return pre

    }
```
