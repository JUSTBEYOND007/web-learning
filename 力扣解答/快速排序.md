var sortArray = function(nums,left=0,right=nums.length-1) {//这样把left和right都传进去

    if(left<right){

        let index=partition(left,right)

        sortArray(nums,left,index-1)//递归调用自身

        sortArray(nums,index+1,right)

    }

    return nums

  

    function partition(left,right){

        let randomIndex=left+Math.floor(Math.random()*(right-left+1));

        [nums[left],nums[randomIndex]]=[nums[randomIndex],nums[left]]

        let i=left+1,j=right

        let privot=nums[left]

        while(i<=j){

            while(i<=j&&nums[i]<privot) i++

            while(i<=j&&nums[j]>privot) j--

            if(i>j) break

            [nums[i],nums[j]]=[nums[j],nums[i]]

            i++

            j--

        }

        [nums[left],nums[j]]=[nums[j],nums[left]]

        return j

    }

};
