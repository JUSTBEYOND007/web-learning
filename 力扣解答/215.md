import java.util.*;

import java.util.Random;

class Solution {

    public static  final Random random=new Random(System.currentTimeMillis());

    public int findKthLargest(int[] nums, int k) {

        int target=nums.length-k;

        int left=0;

        int right=nums.length-1;

        while (true){

            int pivotIndex=partition(nums,left,right);

            if(pivotIndex==target)

                return nums[pivotIndex];

            else if (pivotIndex<target) {

                left=pivotIndex+1;

            } else if (pivotIndex>target) {

                right=pivotIndex-1;

            }

        }

    }

  

    private int partition(int[] nums, int left, int right) {

       int randomInt=left+random.nextInt(right-left+1);

      swap(nums,left,randomInt);//这两行代码的作用是随机选择基准元素pivot

       int pivot=nums[left];

       int le=left+1;

       int ge=right;

       while (true){

           while (le<=ge&&nums[le]<pivot)

               le++;

          while (le<=ge&&nums[ge]>pivot)

               ge--;

           if(le>=ge)

               break;

  

           swap(nums,le,ge);

           le++;

           ge--;

       }

       swap(nums,ge,left);//注意交换的是ge  如果le和ge相等 均可，如果ge<le那么应该是ge

       return ge;//返回的是ge

    }

  

    private void swap(int[] nums, int left, int randomInt) {

        int temp=0;

        temp=nums[left];

        nums[left]=nums[randomInt];

        nums[randomInt]=temp;

    }

}

### 为什么要随机化？

普通快速排序通常选择第一个或最后一个元素作为基准，这在以下情况会导致**性能退化**：


- **数组已排序**：无论升序或降序，每次递归只能将问题规模减少 1（时间复杂度退化为 O (n²)）。
- **大量重复元素**：划分不均匀，递归树高度接近 n。
