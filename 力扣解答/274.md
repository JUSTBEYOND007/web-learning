//数组中有h个不小于h的值，求最大的h

  

class Solution {

    public int hIndex(int[] citations) {

        if (citations==null)

            return 0;

        //先排序 时间复杂度O（nlogn）

        Arrays.sort(citations);

        for ( int i =0; i <=citations.length-1; i++) {//i来表示可能的结果

            int maxH=citations.length-i;//重点

            if(citations[i]>=maxH)

                return maxH;

        }

        return  0;

    }

}
