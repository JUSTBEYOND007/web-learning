递归 
主要是一些判断语句的问题
```js
var sumNumbers = function(root) {

    //dfs+数组记录

    const num=[]

  

    function dfs(root,cur=''){

        if(root===null)//到null节点了  我们不需要num.push(cur)的 因为不是叶子节点

            return

        if(root.left===null&&root.right===null){//这里判断是否是叶子节点

            cur+=root.val

            return num.push(cur)

        }

        cur+=root.val

        dfs(root.left,cur)

        dfs(root.right,cur)

    }

    dfs(root)

    return num.map(Number).reduce((pre,cur)=>pre+cur,0)

};
```
