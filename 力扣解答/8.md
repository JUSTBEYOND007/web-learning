//32位系统 不能使用long等类型 注意-1*(Integer.MIN_VALUE%10))  要用（） 最后再*-1 否侧存在边界问题 Integer.MIN_VALUE绝对值比

//Integer.MAX_VALUE大一

class Solution {

    public int myAtoi(String s) {

        if (s.equals("")||s.equals("-"))

            return 0;

        char[] chars = s.toCharArray();

        int res=0;

        int index=0;//从0开始索引

        while (index<s.length()&&chars[index]==' '){

            index++;

        }

        if(index==s.length())

            return 0;

        int flag=1;

        if(index<s.length()&&chars[index]=='-'){

            flag=-1;

            index++;

        }

        else if(index<s.length()&&chars[index]=='+'){

            index++;

        }

  

        for (int i = index; i < s.length(); i++) {

            if(chars[i]-'0'>=0&&chars[i]-'0'<=9){

                if(res>Integer.MAX_VALUE/10||(res==Integer.MAX_VALUE/10&&chars[i]-'0'>Integer.MAX_VALUE%10))

                    return Integer.MAX_VALUE;

                else if(res<Integer.MIN_VALUE/10||(res==Integer.MIN_VALUE/10&&chars[i]-'0'>-1*(Integer.MIN_VALUE%10)))

                    return Integer.MIN_VALUE;

                res=res*10+flag*(chars[i]-'0');

            }

            else break;

        }

        return res;

    }

}
