function backtrack(参数...) {

    // 1. 终止条件（找到一个答案）
    if (满足结束条件) {
        result.push([...path])  // 收集结果（拷贝）
        return
    }

    // 2. 遍历“当前层”的所有选择
    for (选择 of 当前所有可选项) {

        // 做选择
        path.push(选择)

        // 标记状态（可选，用于去重、used 数组等）
        used[选择] = true   // 如果需要

        // 进入下一层
        backtrack(下一个状态)

        // 撤销选择（回溯）
        path.pop()
        used[选择] = false  // 如果需要
    }
}
