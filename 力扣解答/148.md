思想 归并排序 先分割为一个一个的链表 然后两两组合 合并有序链表 时间空间复杂度O(nlogn)

```
var sortList = function(head) {
    if (!head || !head.next) return head

    // 1. 快慢指针分割链表
    let slow = head
    let fast = head.next

    while (fast && fast.next) {
        slow = slow.next
        fast = fast.next.next
    }

    let mid = slow.next
    slow.next = null//这样来截断 左右链表

    // 2. 递归排序 分割链表
    let left = sortList(head)//返回排序后的头部节点
    let right = sortList(mid)

    // 3. 合并有序链表 
    return mergeList(left, right)
}

function mergeList(l1, l2) {
    const dummy = new ListNode(-1)
    let cur = dummy

    while (l1 && l2) {
        if (l1.val <= l2.val) {
            cur.next = l1
            l1 = l1.next
        } else {
            cur.next = l2
            l2 = l2.next
        }
        cur = cur.next
    }

    cur.next = l1 || l2
    return dummy.next
}
```
