我们 不能选每次跳最远 我们每次都维护一个区间[left,right] 此时我们能跳到的区间
如果这一跳的区间right>=nums.length-1 说明能跳到最后 返回step
思路就是使用[left,right]来维护我们当前跳次数下 能跳到的最远位置 如果覆盖 即返回
```js
var jump = function(nums) {

    let left=0

    let right=0

    let step=0

    let rightMax=0//当前跳能跳到的最远位置

    while(right<nums.length-1){//只要跳不到 就循环

        for (let index = left; index <=right; index++) {

            rightMax=Math.max(rightMax,index+nums[index])

        }

        left=right+1//重要 题上说可以保证可以达到最后 那么这一跳肯定能跳到right+1的位置

        right=rightMax//更新能跳到的区间

        step++

    }

    return step

};
```
