`str = arr[i] + str` 和 `str = str + arr[i]` 的拼接顺序 **完全相反**
```js
var thousandSeparator = function(n) {

    //倒序实现比较简单

    //1234567 的长度是 7，前面只有 1 是不足三位的第一段。如果从前往后，你要写额外逻辑来处理 “第一段长度可能不足三位” 的情况

    let str=''

    let arr=n.toString().split('')//这样来转为数组

    let curLength=0

    for(let i=arr.length-1;i>=0;i--){

        if(curLength<=3){

            str=arr[i]+str

            curLength++

        }else{

            str='.'+str

            str=arr[i]+str

            curLength=1//这里已经填上一位了

        }

    }

    return str

};
```
