# `slice(start, end)` 的规则 左闭右开

1. **start** —— 起始索引（包含该位置的元素）
    
2. **end** —— 结束索引（不包含该位置的元素）
    
3. **负数索引**表示从末尾开始数：-1 是最后一个元素，-2 是倒数第二个，以此类推


```js


var letterCombinations = function(digits) {

    //map保存数字和字母对应关系 回溯解答

    const map=new Map()

    map.set('2',['a','b','c'])

    map.set('3',['d','e','f'])

    map.set('4',['g','h','i'])

    map.set('5',['j','k','l'])

    map.set('6',['m','n','o'])

    map.set('7',['p','q','r','s'])

    map.set('8',['t','u','v'])

    map.set('9',['w','x','y','z'])


    let path=''//修改内容 字符串地址会变 我们用let

    const result=[]

    function traceback(index){

        if(index===digits.length)

            return result.push(new String(path))

        for (let i = 0; i < map.get(digits.charAt(index)).length; i++) {

            const ch=map.get(digits.charAt(index))[i]

            path=path+ch

            traceback(index+1)

            path=path.slice(0,-1)//！删除最后一个字符 我们可以用-1来表示最后一个元素
	            //含头不含尾部

        }

    }

  

    traceback(0)

    return result

};
```
