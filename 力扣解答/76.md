题中“涵盖 `t` 所有字符的最小子串”
那么我们就用<font color="Blue">哈希表</font>存储 对应字符 的出现次数 和总数量 只要别的字串都都满足这两个需求就符合 “涵盖 `t` 所有字符的最小子串”





```
import java.util.HashMap;

  

class Solution {

    public String minWindow(String s, String t) {

        HashMap<Character,Integer> windows_map=new HashMap<>();

        HashMap<Character,Integer> t_map=new HashMap<>();

        if(s.length()==0||t.length()==0)

            return "";

        for (int i = 0; i < t.length(); i++) {

            t_map.put(t.charAt(i),t_map.getOrDefault(t.charAt(i),0)+1);

        }

        int count=t_map.size();//获得t字串中字符数量

        int have=0;//s字串中包含对应t字串中字符对应数量的个数

        int left=0,right=0;

        int resstart=0,resLength= Integer.MAX_VALUE;

        while (right<=s.length()-1){

            if(t_map.containsKey(s.charAt(right))){

                windows_map.put(s.charAt(right),windows_map.getOrDefault(s.charAt(right),0)+1);

                if(windows_map.get(s.charAt(right)).equals(t_map.get(s.charAt(right)).intValue()))

                    have++;

            }

            while (have==count&&left<=right){

                if ((right-left+1)<resLength){

                    resLength=right-left+1;

                    resstart=left;

                }

                if(t_map.containsKey(s.charAt(left))){

                    if(windows_map.get(s.charAt(left)).equals(t_map.get(s.charAt(left)))) {

                        have--;

                    }

                    windows_map.put(s.charAt(left),windows_map.get(s.charAt(left))-1);

                }

                left++;//不管是不是t中元素 都要++    以此位置为开头的位置的字串此时已经是最短了，更换左端点 寻找别的符合条件的  

            }

            right++;

        }

  

        return  resLength==Integer.MAX_VALUE? "" :s.substring(resstart,resstart+resLength);

    }

}
```
