相同递归的套路
但是function(root) 是一个变量 我们需要传左右子树 所以创建一个新的函数

```js
var isSymmetric = function(root) {

    function fn (left,right){

        if(!left&&!right)  //也是先判断底层 如果左右均为空 为true

            return true

        else if(!left||!right)  //前面先判断了是否同时为null 所以这里就是排除其中一个为null的情况

            return false

        return left.val==right.val&&fn(left.left,right.right)&&fn(left.right,right.left)//递归和判断 当前值 返回的结果为三者的结合 一个为false 即为false  所以我们用&&

    }

    return fn(root.left,root.right)

};
```
