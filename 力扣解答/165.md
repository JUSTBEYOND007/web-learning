我们通过split分割字符串 获得字符数组
注意 
version1 =
"1.0"
version2 =
"1.0.0.0"
这两一样
我们要 lens=Math.max(arr1.length,arr2.length) 这样我们两者比较完 如果长度不够 我们去认为为0

这里1.1 和1.01 解析  01在前导 0 在转成 Number 类型时会被忽略
这里不需要怕

```js
var compareVersion = function(version1, version2) {

    const arr1=version1.split('.')

    const arr2=version2.split('.')

    let lens=Math.max(arr1.length,arr2.length)

    for(let i=0;i<lens;i++){

        let num1=i<arr1.length?Number(arr1[i]):0//要判断！

        let num2=i<arr2.length?Number(arr2[i]):0

        if(num1>num2)

            return 1

        else if(num1==num2){

            continue

        }else return -1

    }

    return 0

};
```
