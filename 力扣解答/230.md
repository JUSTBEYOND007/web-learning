中序遍历 闭包count计数 
```js
var kthSmallest = function(root, k) {

    //中序遍历 闭包count

    let count=0,ans=0//ans为结果

    function fn(root){

        if(!root){

            return false //false表示不是第k个

        }

        if(fn(root.left)){

            return true

        }

         count++

        if(count===k){

            ans=root.val

            return true

        }

        if(fn(root.right)){

            return true

        }

    }

    fn(root)

    return ans

};
```
