class Solution {

    public double findMedianSortedArrays(int[] nums1, int[] nums2) {

        int m=nums1.length;

        int n=nums2.length;

        if(m>n)

            return findMedianSortedArrays(nums2,nums1);//目的是让nums1为较短的一
//个 ，这样复杂度为O(min(m,n))  因为我们是在较短的数组上二分查找目标的位置i 
        int left=0,right=m;

        int leftSize=(m+n+1)/2;

        while (left<=right){

            int i=(left+right)/2;

            int j=leftSize-i;//由leftSize（表示总数的一半或者一半加1） 这样可以
            //j的位置

  

            int left1=i==0? Integer.MIN_VALUE:nums1[i-1];//边界条件处理 就是
            //重组后 左侧元素 全是num1或者nums2的元素的情况 

            int right1=i==m? Integer.MAX_VALUE:nums1[i];

            int left2=j==0? Integer.MIN_VALUE:nums2[j-1];

            int right2=j==n? Integer.MAX_VALUE:nums2[j];

  

            if(left1<=right2&&left2<=right1){

                if((m+n)%2==1)

                    return Math.max(left1,left2);

                else

                    return (Math.max(left1,left2)+Math.min(right1,right2))/2.0;

            }

            else if(left1>right2)//说明i所在位置太靠右了

                right=i-1;//查找左侧元素

            else

                left=i+1;

        }

        return  -1;//不应该到达这里 只是为了符合语法

    }

}
