不太会的题
遍历字符串时，将「括号前的字符串」和「重复次数」作为一组上下文暂存到栈中；
遇到闭合括号时，弹出栈顶的上下文，将当前括号内的字符串重复指定次数后，与前缀字符串拼接；
处理多位数的重复次数（如 100[a] 中的 100），需通过累积的方式计算，而非直接取单个数字。

```js
var decodeString = function(s) {

    const stack=[]

    let res=''

    let k=0

    for (const element of s) {

        if(element==='['){

            stack.push([res,k])

            res=''

            k=0

        }else if(element>='a'&&element<='z'){

            res+=element

        }else if(element>='0'&&element<='9'){

            k=k*10+parseInt(element)//用parseInt来解析为number类型

        }else{

            const [pre_res,pre_k]=stack.pop()

            res=pre_res+res.repeat(pre_k)//用repeat自带方法 来重复pre_k次

        }

    }

    return res

};
```
