括号有效  ，即保证
- 左括号数量大于右括号数量
- 左括号数量小于n （因为我们是从0开始计数的）
即可


```js
var generateParenthesis = function(n) {

    const path=[]

    const result=[]
	//left,right是指左右括号的个数
    function traceback(left,right){

        if(left===n&&right===n){

            result.push(path.join(''))

        }

        if(left<n){

            path.push('(')

            traceback(left+1,right)

            path.pop()

        }

        if(right<left){

            path.push(')')

            traceback(left,right+1)

            path.pop()//这里也需要pop来恢复状态

        }

    }

    traceback(0,0)

    return result

};
```
