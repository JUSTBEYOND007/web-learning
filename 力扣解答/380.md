class RandomizedSet {

    //哈希＋数组

    private Map<Integer,Integer> randomizedSet;

    private ArrayList<Integer> list;

    public RandomizedSet() {

         randomizedSet=new HashMap<>();

         list=new ArrayList<>();

    }

  

    public boolean insert(int val) {

        boolean add = randomizedSet.containsKey(val);

        if (!add) {

            randomizedSet.put(val,randomizedSet.size());

            list.add(val);

        }

        return !add;

    }

  

    public boolean remove(int val) {

        boolean remove = randomizedSet.containsKey(val);

  /*删除操作时，首先判断 val 是否在哈希表中，如果不存在则返回 false，如果存在则删除 val，操作如下：

从哈希表中获得 val 的下标 index；

将变长数组的最后一个元素 last 移动到下标 index 处，在哈希表中将 last 的下标更新为 index；

在变长数组中删除最后一个元素，在哈希表中删除 val；

返回 true

作者：力扣官方题解
链接：https://leetcode.cn/problems/insert-delete-getrandom-o1/solutions/1411578/o1-shi-jian-cha-ru-shan-chu-he-huo-qu-su-rlz2/

  */

        if (remove) {

            list.set(randomizedSet.get(val),list.get(list.size()-1));

            randomizedSet.put(list.get(list.size()-1),randomizedSet.get(val));

            list.remove(list.size()-1);//将最后一个

            randomizedSet.remove(val);

        }

        return  remove;

    }

  

    public int getRandom() {

        Random random=new Random();

        return list.get(random.nextInt(list.size()));

    }

}
