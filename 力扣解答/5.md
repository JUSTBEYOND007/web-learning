这是中心扩散算法 从中心向两边扩散  每个点都可能是中心 所以我们需要for循环 遍历所有索引作为中心  同时有奇数和偶数 两种中心方法  我们可以抽象成一个方法 
```js
var longestPalindrome = function(s) {

    let maxLen=0,max_start=0

  

    const expand=(start,end)=>{
		//必须先判断是否越界再判断字符是否相等
        while(start>=0&&end<=s.length&&s.charAt(start)===s.charAt(end)){

            if(end-start+1>maxLen){

                maxLen=end-start+1

                max_start=start

            }

            start--

            end++

        }

    }

    for(let i=0;i<s.length;i++){

        expand(i,i)

        expand(i,i+1)//偶数 我们是(i,i+1)

    }

    return s.substring(max_start,max_start+maxLen)

};
```

第二种方法 是dp rows表示开始 cols表示结束
因为dp[rows]/[cols] 和dp[rows+1]/[cols-1] 有关
空间上 dp[rows+1]/[cols-1] 在dp[rows]/[cols] 左下角 我们需要先更新dp[rows+1]/[cols-1] 才能正确保证dp[rows]/[cols]    所以我们先遍历列再遍历行
同时 cols-rows<3  无论是相等 （同一位置）还是差值为1 （偶数对称）还是为2（中间有一个值，奇数对称）
我们直接赋值为true即可
```js
var longestPalindrome = function(s) {

    const dp=Array.from({length:s.length},()=>new Array(s.length).fill(false))

    let maxLen=0,start=0,end=0

    for (let cols = 0; cols < s.length; cols++) {

        for (let rows = 0; rows <= cols; rows++) {//开头rows要小于cols

            if(s.charAt(cols)===s.charAt(rows)){

                if(cols-rows<3){

                    dp[rows][cols]=true

                }

                else dp[rows][cols]=dp[rows+1][cols-1]

                if(dp[rows][cols]){

                    if(cols-rows+1>maxLen){

                        start=rows

                        end=cols

                        maxLen=cols-rows+1

                    }

                }    

            }

            else dp[rows][cols]==false

        }

    }

    return s.substring(start,end+1)

};
```

```java
class Solution {  
    public String longestPalindrome(String s) {  
        int res_start=0,res_end=0;  
        int res=0;  
        for (int i = 0; i < s.length() - 1; i++) {  
        //我们这种方法是从一个点 向外发散型
            //对于bab    
            int L=i,R=i; //起始位置相同 
            while (L>=0&&R<=s.length()-1&&s.charAt(L)==s.charAt(R)){  
                if(res<(R-L+1)){  
                    res=R-L+1;  
                    res_start=L;  
                    res_end=R;  
                }  
                L--;  
                R++;   //我们这种方法是从一个点 向外发散型
            }  
            //对于baab这种类型  //有两种回文子串类型 都要遍历判断下
            L=i;  
            R=i+1;  //R初始为L+1

                if(res<(R-L+1)){  
                    res=R-L+1;  
                    res_start=L;  
                    res_end=R;  
                }  
                L--;  
                R++;  
            }  
        }  
        return s.substring(res_start,res_end+1);
```

```java
class Solution {

    public String longestPalindrome(String s) {

        boolean dp[][]=new boolean[s.length()][s.length()];

        for (int i = 0; i < s.length() ; i++) {

            dp[i][i]=true;

        }

        if (s.length()<2) {

            return s;

        }

        int res=0;

        int res_start=0,res_end=0;

	//由递归式子 若s.charAt(row)!=s.charAt(col)  dp[i][j] 和dp[i+1][j-1] 关系
	由左下角dp[i+1][j-1]知道 dp[i][j] 的值  所以我们按照列来遍历
        for (int col = 1; col < s.length(); col++) {//col表示列 表示右侧

            for (int row = 0; row < col ; row++) {//row表示左

                if (s.charAt(row)!=s.charAt(col)) {

                    dp[row][col]=false;

                    continue;

                }

                if (col-row<3)

                    dp[row][col]=true;

                else if (dp[row+1][col-1]) {

                    dp[row][col]=true;

                }

                if (dp[row][col]&&col-row+1>res){

                    res=col-row+1;

                    res_start=row;

                    res_end=col;

                }

            }

        }

        return s.substring(res_start,res_end+1);

    }

}
```
