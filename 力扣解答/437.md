思路和力扣560 比较类似 使用前缀和  但是这里是树 还需要回溯恢复map状态 因为只能是从父节点到子节点
用pre表示前缀和 map表示值和数量



```js
var pathSum = function(root, targetSum) {

    const map=new Map()

    map.set(0,1)//注意初始化 

    let count=0


    function dfs(root,pre){

       if(!root)

            return null

        pre+=root.val

        if(map.has(pre-targetSum)){          

            count+=map.get(pre-targetSum)

        }//先计算count再更新map 因为map统计的是前面的路径的值 当前节点之前已经经过的路径和

        map.set(pre,(map.get(pre)||0)+1)//(map.get(pre)||0)

        dfs(root.left,pre)

        dfs(root.right,pre)

        map.set(pre,(map.get(pre)-1))//再从map中删除掉这条记录

    }

    dfs(root,0)

    return count

};
```

这里我们使用map.set(pre,(map.get(pre)||0)+1) 来赋值
对于map.get(pre)如果不存在 返回的是undefined  为false 会执行0+1
