判断岛屿个数   我们就深度优先遍历该岛屿的上下左右是否为1 如果为1我们就接着遍历 
直到超出边界或者不为1
这道题<font color="#2E8B57">默认是可以直接修改原数组的，标准答案也是这样做的，如果访问过我们直接设置为0</font>
```js
var numIslands = function(grid) {

    if(!grid||grid.length==0)

        return

    const rows=grid.length,cols=grid[0].length

    let count=0

    for(let i=0;i<rows;i++){

        for(let j=0;j<cols;j++){

            if(grid[i][j]==='1'){

                count++

                dfs(i,j)

            }

        }

    }

    function dfs(curRow,curCol){

        if(curRow<0||curRow>=rows||curCol<0||curCol>=cols||grid[curRow][curCol]==='0')

            return

        grid[curRow][curCol]='0'

        dfs(curRow-1,curCol)

        dfs(curRow,curCol-1)

        dfs(curRow,curCol+1)

        dfs(curRow+1,curCol)
    }
    return count

};
```
