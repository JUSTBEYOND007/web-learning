import java.util.ArrayList;  
import java.util.Collections;  
  
/**  
 * Definition for singly-linked list. * public class ListNode { *     int val; *     ListNode next; *     ListNode() {} *     ListNode(int val) { this.val = val; } *     ListNode(int val, ListNode next) { this.val = val; this.next = next; } * } */class Solution {  
    public ListNode mergeKLists(ListNode[] lists) {  
        if (lists.length==0) {  
            return  null;  
        }  
        while (lists.length>1){  //集合来存储两两链表合并后的链表
            ArrayList<ListNode> mergeList=new ArrayList<>();  
            for (int i=0;i<lists.length;i+=2){  //注意是i+=2 走俩步 使其恒为两两链表合并的左侧链表
                ListNode list1=lists[i];  
                ListNode list2=null;  
                if(i+1<=lists.length-1){  
                    list2=lists[i+1];  //判断下 因为奇数个最后一个链表可能为空
                }  
                ListNode listNode = mergeTwoList(list1, list2);  
                mergeList.add(listNode);  
            }  
            ListNode nowLists[]=new ListNode[mergeList.size()];  
            for (int j = 0; j < mergeList.size(); j++) {  
                nowLists[j]=mergeList.get(j); //复制下集合中的值给数组lists 
            }  
            lists=nowLists;  
        }  
        return lists[0];  
    }  //mergeTwoList合并两个链表  
    public  ListNode mergeTwoList(ListNode list1,ListNode list2){  
        ListNode dummy=new ListNode(0,null);  
        ListNode cur=dummy;  
        while (list1!=null&&list2!=null){  
            if (list1.val<list2.val) {  
                cur.next=list1;  
                cur=list1;  
                list1=list1.next;  
            }else {  
                cur.next=list2;  
                cur=list2;  
                list2=list2.next;  
            }  
        }  
        cur.next=list1==null? list2:list1;//这里很巧 只需要判断cur1为不为空即可  也不用再遍历 直接cur.next为   cur1或者cur2 链接起来即可
        return dummy.next;  
    }  
}
