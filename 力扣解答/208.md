其实这里 前缀树 就是一个森林 里面有很多树 每个树都是多叉树 node=node[ch]就是向下遍历node[ch]={}就是创建ch子节点 

题较简单的 但是要知道 我们把单词分为每个字母 作为一个节点
每个节点是一个对象 {} 来保存指向后面字母节点的地址 并且结尾字母有isEnd 作为是否是结尾即可 来表示是否含有该单词
整体 主要过程就是遍历即可

```js
var Trie = function() {

    this.children={}//根节点不保存东西 isEnd属性 只需要在最后的字母出现 所以我们不在这里创建

};

/**

 * @param {string} word

 * @return {void}

 */

Trie.prototype.insert = function(word) {

    let node=this.children//根节点不保存东西 this.children指向第一个包含字母的节点

    for (const ch of word) {//就是遍历 是否存在

        if(!node[ch]){

            node[ch]={}//不存在则新建

        }

        node=node[ch]//接着指向子节点

    }

    node.isEnd=true

};

  

/**

 * @param {string} word

 * @return {boolean}

 */

Trie.prototype.search = function(word) {

    let node=this.children

    for (const ch of word) {//就是遍历 逻辑类似

        if(!node[ch]){

            return false

        }

        node=node[ch]

    }

    return node.isEnd?node.isEnd:false//最后一个元素我们判断isEnd

};

  

/**

 * @param {string} prefix

 * @return {boolean}

 */

Trie.prototype.startsWith = function(prefix) {

    let node=this.children

    for (const ch of prefix) {//就是遍历 逻辑类似

        if(!node[ch]){

            return false

        }

        node=node[ch]

    }

    return true//这里只需要判断每个字母是否按顺序在前缀树中包含即可

};

  

/**

 * Your Trie object will be instantiated and called as such:

 * var obj = new Trie()

 * obj.insert(word)

 * var param_2 = obj.search(word)

 * var param_3 = obj.startsWith(prefix)

 */
```
