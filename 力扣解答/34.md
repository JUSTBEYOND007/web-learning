二分查找 原来是找到对应的index  即返回
这里我们找到对应的index但是可能不是左右边界 我们继续修改left和right来查找左右边界
还是直到不满足left<=right条件时停止

题和思路较简单  代码较长但不复杂
 Array.of (值，值)  创建新数组！
```js

var searchRange = function(nums, target) {

    //两次二分查找 分别查找左右边界

    let left=0,right=nums.length-1

    let leftIndex=-1,rightIndex=-1//我们直接初始化为-1 这样不满足也不需要判断返回即可
	///查找左边界
    while(left<=right){

        let mid=Math.floor((left+right)/2)//注意用Math.floor！

        if(nums[mid]===target){

            leftIndex=mid

            right=mid-1//查找左边界 我们继续修改right在左侧查找

        }

        else if(nums[mid]<target){

            left=mid+1

        }else right=mid-1

    }

    left=0,right=nums.length-1//恢复到左右边界！！！
///查找右边界
    while(left<=right){

        let mid=Math.floor((left+right)/2)

        if(nums[mid]===target){

            rightIndex=mid

            left=mid+1//查找右边界 我们继续修改right在左侧查找

        }

        else if(nums[mid]<target){

            left=mid+1

        }else right=mid-1

    }

    return Array.of(leftIndex,rightIndex)

};
```

两次二分查找函数重复 我们可以用isLeft还判断查找左边界还是右边界 抽象成函数
```js

var searchRange = function(nums, target) {

    //两次二分查找

    let leftIndex=-1,rightIndex=-1

    leftIndex=binarySearch(0,nums.length-1,true)

    rightIndex=binarySearch(0,nums.length-1,false)

  

    return Array.of(leftIndex,rightIndex)

    function binarySearch(left,right,isLeft){

        let index=0//记录边界

        while(left<=right){

            let mid=Math.floor((left+right)/2)

            if(nums[mid]===target){

                index=mid//都要记录index

                if(isLeft){  

                    right=mid-1

                }else{

                    left=mid+1

                }

            }

            else if(nums[mid]<target){

                left=mid+1

            }else right=mid-1

        }

        return nums[index]===target?index:-1//可能没有找到 我们判断下

    }

};
```
