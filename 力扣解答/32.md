import java.util.ArrayDeque;

  
//方法 
class Solution {

    public int longestValidParentheses(String s) {

        if (s=="")

            return 0;

        ArrayDeque<Integer> stack=new ArrayDeque<>();

        Boolean flag[]=new Boolean[s.length()];//用数组来存储当前字符能否匹配上
      /*   因为你使用了包装类型 `Boolean` 而非基本类型 `boolean`，导致数组元素默认初始值为 `null`，而非 `false`。  
 我们需要 Arrays.fill(flag,false);否则就报空指针异常 或者我们用  
 boolean flag[]=new boolean[s.length()]; */
        for (int i = 0; i < s.length(); i++) {

            if(s.charAt(i)=='(')

                stack.push(i);//遍历一遍 (则压入 )如果栈中有（则弹出

            else {

                if(!stack.isEmpty()){//即为(

                    Integer index = stack.poll();

                    flag[index]=true;

                    flag[i]=true;

                }

            }

        }

        int curLen=0,resLen=0;

        for (int i = 0; i < flag.length; i++) {

            if (flag[i]){

                curLen++;//表示当前位置的符合题意的字串长度

                resLen=Math.max(curLen,resLen);

            }else {

                curLen=0;

            }

        }

        return  resLen;

    }

}
