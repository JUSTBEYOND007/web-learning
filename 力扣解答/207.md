其实就是判断拓扑排序 无环  就可以完成所有课程的学习

解决拓扑排序的方法

- **统计入度**：`inagree[course]++`
    
    - 表示每门课程还有多少前置课程未修。
        
- **建图存出度**：`map.get(pre).push(course)`
    
    - 存储每门课程作为前置的后续课程列表。
        
- **初始化队列**：所有入度为 0 的课程入队
    
    - 这些课程可以直接修，不依赖其他课程。
        
- **BFS处理队列**：
    
    - 取出队首课程，算作“已完成”，`count++`
        
    - 遍历它的后续课程，入度减 1
        
    - 如果后续课程入度变为 0 → 入队
        
- **判环**：
    
    - BFS 结束后，如果 `count === numCourses` → 无环 → 可以完成所有课程 → 返回 `true`
        
    - 否则 → 有环 → 无法完成所有课程 → 返回 `false`


```js
var canFinish = function(numCourses, prerequisites) {

    //bfs解决拓扑排序问题

    const map=new Map()//记录出度

    const inagree=new Array(numCourses).fill(0)//记录入读个数
	//我们数组必须初始化赋值fill(0) 否则为undefined 会报错
    for(let i=0;i<prerequisites.length;i++){

        const [course,pre]=prerequisites[i]

        inagree[course]++

        // map.set(prerequisites[i][1],[...map.get(prerequisites[i][1]),prerequisites[i][0]])

        if(!map.get(pre)){

            map.set(pre,[])//这里我们首先判断下 使得value为数组

        }

        map.get(pre).push(course)

    }

    const queue=[]

    let count=0

    for(let i=0;i<inagree.length;i++){

        if(inagree[i]===0){

             queue.push(i)

             count++ //说明该课程可以执行

        }

    }

    while(queue.length>0){

        const node=queue.shift()

        if(!map.get(node)) continue//可能为空 没有后续课程 我们直接continue

        for(let next of map.get(node)){

            inagree[next]--

            if(inagree[next]===0){

                queue.push(next)

                count++

            }

        }

    }

    return count===numCourses//这样直接判断是否相等即返回true或者false了

};
```
