我们考研的时候学过 用手会构造 思路一致  问题就在于代码如何实现

先序遍历我们可以确定根节点 题目树节点值唯一 我们可以使用map来记录值和下标 方便查找根节点在中序遍历中的位置
我们可以根据中序遍历获得左右子树的大小  map.get(preorder[preL])-inL
然后深度递归左 右子树 构造节点返回

区间我们使用左闭右开 是因为：

1️⃣ 让「长度计算」永远简单  
2️⃣ 让「空区间判断」只用一个条件  
3️⃣ 让「递归切分」几乎不需要纠结 +1 / -1  
4️⃣ 大幅降低你现在最痛苦的「索引迷惑」


```
var buildTree = function(preorder, inorder) {

    const map=new Map()

    if(!inorder||!preorder)

        return null

    for(let i=0;i<inorder.length;i++){

        map.set(inorder[i],i)

    }
	//preL,preR,inL分别表示子树在preorder的左边界（根节点位置）、有边界 是左闭右开的！inorder的左边界
    function dfs(preL,preR,inL){

        if(preL===preR)

            return null //说明无元素 返回Null

        let leftSize=map.get(preorder[preL])-inL 
        //map.get(preorder[preL])获得根节点在inorder的位置 减去左边界获得左子树的长度
        let left=dfs(preL+1,preL+1+leftSize,inL)
//preL为根节点位置 preL+1为左子树开始位置 preL+1+leftSize 左子树右边界，inL仍为inorder的左边界
        let right=dfs(preL+1+leftSize,preR,map.get(preorder[preL])+1)
//preL+1+leftSize 在preorder中左子树右边界即右子树开始，preR仍为有边界,map.get(preorder[preL])为inorder根节点 +1即为右子树开始位置
        return new  TreeNode(preorder[preL],left,right)

    }

     return dfs(0,inorder.length,0)

};
```
