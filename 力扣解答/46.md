简单题
重点是我们要知道要用used 来保存该元素是否使用过
path来表示当前路径下的元素数组

数组中每一个元素都可以作为当前元素加入  用for循环 遍历所有情况

回溯函数不需要参数  path是外部变量
```js
/**

 * @param {number[]} nums

 * @return {number[][]}

 */

var permute = function(nums) {

    //使用used判断是否已经加入

    const result=[]

    const used=new Array(nums.length).fill(false)//记得填充长度nums.length

    const path=[]

  

    function fallback(){

        if(path.length===nums.length)

            result.push([...path])//存结果（注意要拷贝） path是引用类型

        for (let i = 0; i < used.length; i++) {//for循环来查找还未加入的

            if(!used[i]){

                path.push(nums[i])

                used[i]=true

                fallback()

                path.pop()

                used[i]=false

            }

        }

    }

    fallback()

    return result

};
```
