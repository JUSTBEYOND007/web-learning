

我们发现 dp只和dp[k-1]和dp[k-2]有关 所以 我们可以用两个变量来表示即可
```js
var climbStairs = function(n) {

    const dp=[]

    dp[1]=1

    dp[2]=2

    for(let k=3;k<=n;k++){

        dp[k]=dp[k-1]+dp[k-2]

    }

    return dp[n]

};
```

```js
var climbStairs = function(n) {

    let l1=1,l2=2

    if(n<=2)

        return n

    let ans=0

    for(let i= 3;i<=n;i++){

        ans=l1+l2

        l1=l2

        l2=ans

    }

    return ans

};
```
