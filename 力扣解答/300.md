在 **子序列（subsequence）类 DP 题** 中，dp 数组的经典定义几乎都是：
# ⭐ **DP 套路：dp[i] 表示以 i 结尾的某种最优子序列的长度**


```js
//dp表示字符串以i结尾的最大子序列长度 （提议我们不需要是以最后元素结尾 所以我们用maxAns记录）
var lengthOfLIS = function(nums) {

    const dp=new Array(nums.length+1).fill(1)//初始化最小都包含自身一个元素 长度为1

    let maxAns=1

    for(let i=0;i<nums.length;i++){

        for(let j=0;j<i;j++){//从前往后遍历 子字符串 更新nums值

            if(nums[i]>nums[j]){//如果大 说明要更新

                dp[i]=Math.max(dp[i],dp[j]+1)//

                maxAns=Math.max(dp[i],maxAns)

            }

        }

    }

    return maxAns
};

```


   ```java
   class Solution {

    public int lengthOfLIS(int[] nums) {

        //以第n个元素结尾的最长递增子序列

        int []dp=new int[nums.length];

        Arrays.fill(dp,1);

        int maxNumber=1;

        for (int i = 0; i < nums.length; i++) {

            for (int j = 0; j < i; j++) {

                if(nums[j]<nums[i]){

                    dp[i]=Math.max(dp[i],dp[j]+1);

                    maxNumber=Math.max(maxNumber,dp[i]);

                }

            }

        }

        return maxNumber;

    }

}
