中序遍历 是指根元素 中间遍历 即按照<font color="#9B59B6">左 根 右</font> 的顺序遍历
一直不会 一生之痛
先if判断条件 如果为null 直接返回  使用闭包  外部res变量 内部dfs函数 调用dfs函数！！！
```
var inorderTraversal = function(root) {

    const res=[]

    function dfs(root){

        if(!root)

            return

        dfs(root.left)

        res.push(root.val)

        dfs(root.right)

    }

    dfs(root)

    return res

};
```
