```js
var uniquePaths = function(m, n) {
	//初始化为1 第一行和第一列值均为1 别的行列用dp更新
    const dp=Array.from({length:m},()=>new Array(n).fill(1))//创建二维数组！！！

    if(m==1||n==1)

        return 1 //边界判断 只有一行或者一列 返回1

    dp[0][1]=1

    dp[1][0]=1

    for(let i=0;i<m;i++){

        for(let j=0;j<n;j++){

            if(i>=1&&j>=1){

                dp[i][j]=dp[i-1][j]+dp[i][j-1]//只能向右或者向下 dp为这样

            }

        }

    }
    return dp[m-1][n-1]
};
```


```java
class Solution {

    public int uniquePaths(int m, int n) {

        int dp[][]=new int[m+1][n+1];
		//这样做的目的是让下标从1开始，方便处理边界，不用担心i-1或j-1越界


       for (int i = 0; i <=m; i++) {

            for (int j = 0; j <=n; j++) {

                dp[i][j]=1;

            }

        }

        for (int i = 2; i <=m; i++) {
        //从第二行第二格开始即可 
        //第一行第二格和第二行第一格初始化为了1

            for (int j = 2; j <=n; j++) {

                dp[i][j]=dp[i-1][j]+dp[i][j-1];

            }

        }

        return  dp[m][n];

    }

}
```

：**机器人只能向下或向右移动**，因此到达第一行或第一列的任何位置都只有**唯一一条路径**。
