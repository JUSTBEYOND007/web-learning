题目要求乘积最大子数组
子数组元素值可能为正也可能为负 所以我们需要维护以索引i结尾的子数组的最大值和最小值imax和imin
因为imax和imin只和前面i-1的索引有关 我们可以只使用变量表示即可
用max来表示乘积最大子数组的结果值
```js
var maxProduct = function(nums) {

    let imax=nums[0],imin=imax,max=imax

    for(let i=1;i<nums.length;i++){

        if(nums[i]<0){
			//重点 如果i下标为负值 那么imax和imin需要交换颠倒
            [imax,imin]=[imin,imax]

        }

        imax=Math.max(imax*nums[i],nums[i])

        imin=Math.min(imin*nums[i],nums[i])

        max=Math.max(max,imax)

    }
    return max
};
```

```java
class Solution {

    public int maxProduct(int[] nums) {
//因为有正负数，所以我们需要保存最小值和最大值 因为负负得正
        int max_cur=nums[0];//子数组是连续的 我们用max_cur、max_cur来保存 当前下表结尾的最大乘积值即可

        int max_cur=nums[0];

        int MaxNum=nums[0];

        for (int i = 1; i < nums.length; i++) {//从1开始即可

            int max_temp=max_cur,min_temp=min_cur;

            max_cur=Math.max(Math.max(max_temp*nums[i],min_temp*nums[i]),nums[i]);
//注意需要和当前nums[i]比较 因为可能从当前位置开始值最大
            min_cur=Math.min(Math.min(max_temp*nums[i],min_temp*nums[i]),nums[i]);

            MaxNum=Math.max(MaxNum,max_cur);

        }

  
  

        return MaxNum;

    }

}
```
