通过先排序 然后 我们递归 index 只能往后遍历 来避免重复
二维数组的去重复 比较困难 不要这样做 “我们最后再去重复” 

先排序剪枝  然后再回溯 这样能优化性能 减少时间 但是时间复杂度是没变

两种情况 1️⃣ 选当前数字    2️⃣ 不选当前数字，去下一个
```js

var combinationSum = function(candidates, target) {

    candidates.sort((a,b)=>a-b)//排序的是 当前数组本身  升序排序

    const result=[]

    const path=[]

    function traceback(index,pre){

        if(pre===target){

            result.push([...path])

            return

        }

        for (let i = index; i < candidates.length; i++) {

            const element = candidates[i];

            if(pre+element>target)

                return

            path.push(element)

            pre+=element

            traceback(i,pre)

            pre-=element

            path.pop()

        }

    }

    traceback(0,0)

    return result

};
```
