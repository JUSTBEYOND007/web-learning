一个整数数组 `nums` ，其中元素已经按 **升序** 排列，请你将其转换为一棵 平衡 二叉搜索树。

方法就是取中间值作为根节点 递归构造左右子树
```js
var sortedArrayToBST = function(nums) {

    function fn(left,right){

        if(left>right)//说明不存在对应的节点 返回null即可

            return null

        let middle=Math.floor((left+right)/2)//记得Math.floor

        const node=new ListNode(nums[middle])//创建节点

        node.left=fn(left,middle-1)//返回的是左侧的中间节点

        node.right=fn(middle+1,right)////返回的是右侧的中间节点

        return node

    }

   return fn(0,nums.length-1)

};
```
