思路中等  代码较长
我们是要找是否能找到  找到我们就return  否则return false
```js
var exist = function(board, word) {

    //图论+回溯

    let flag=false

    // const used=new Array(board).fill(false)//这样new二维数组可以么？

    const used=Array.from({length:board.length},()=>Array(board[0].length).fill(false))

    for (let i = 0; i < board.length; i++) {

        for (let j = 0; j < board[0].length; j++) {

            if(board[i][j]===word[0]){//我们先找到第一个入手点

                used[i][j]=true

                if(traceback(i,j,1)){//传入坐标和1表示接下来找和word[1]一样的元素位置

                    return true//及时结束回溯

                }

                used[i][j]=false

            }

        }    

    }

    function traceback(row,col,index){

        if(index===word.length)//说明全都找到了

            return true

        const directions=[[1,0],[-1,0],[0,1],[0,-1]]

        for(const [dx,dy] of directions){

            let curRow=row+dx

            let curCol=col+dy

            if(curRow>=0&&curRow<board.length&&curCol>=0&&curCol<board[0].length&&used[curRow][curCol]===false&&board[curRow][curCol]===word[index]){

                used[curRow][curCol]=true

                if(traceback(curRow,curCol,index+1)) return true
		
                used[curRow][curCol]=false

            }

        }

        return false//没有执行前面的return true 说明不行就return false

    }

    return flag

};
```
