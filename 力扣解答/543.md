我们需要保存闭包ans变量 来保存最大值
二叉树的直径 就是左右子树的深度相加 即当前节点的直径（不是二叉树的直径）和 左右子树中节点的最大直径 取最大值

```js
var diameterOfBinaryTree = function(root) {
    let ans = 0; // 用于记录最大直径

    function fn(root) {
        if(root === null)
            return 0; // 空节点返回高度0
        let l1 = fn(root.left);   // 左子树高度
        let l2 = fn(root.right);  // 右子树高度
        ans = Math.max(ans, l1 + l2); // 更新最大直径（路径长度 = 左高 + 右高）
        //l1 + l2是因为边数=深度-1  所以+2-1-1 =0了
        return Math.max(l1, l2) + 1;  // 返回当前节点的高度
    }

    fn(root);
    return ans; // 返回二叉树的直径
};

```
