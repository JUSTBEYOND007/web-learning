```js
class Solution {  
    public int canCompleteCircuit(int[] gas, int[] cost) {  
        //遍历从每个加油站开始环绕一圈  
        for (int i = 0; i < gas.length; ) {  
            int step=0,cur=0;  
            while (step<cost.length) {  
                int k=(i+step)%gas.length;  k表示的当前索引
                cur+=gas[k]-cost[k];  
                if(cur<0) break;  
                step++;  
            }  
            if(step==cost.length)  
                return i;  
            i=i+step+1;  //表示下一个可能是答案的起始位置
        }  
        return -1;  
    }  
}
时间O（n） 空间O（1）
```
