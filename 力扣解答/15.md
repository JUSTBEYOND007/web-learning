双指针 加sort去除重复元素

首先排序 并且每一步我们判断和后面的元素是否一样 来保证结果无重复的

三个元素 我们for循环 用i来指定第一个元素 l r指向后面数组的首尾 向中间遍历 O（n） 外部for循环O(n) 总共O(n平方)   
三个元素我们固定一个 就变成两数之和为一固定值了（不用两数之和中的map 是为了节省空间复杂度
两数之和题 但是返回的是角标 和此题不同 所以不用双指针）


是一个非常经典且让许多初学者（甚至资深开发者）头疼的问题。简单来说，这是因为 JavaScript 设计之初，`Array.prototype.sort()` 的默认行为是**基于字符串 Unicode 码点（UTF-16 code units）排序**的，而不是基于数值大小。

以下是深层原因和逻辑解析：

### 1. 默认的转换机制

当你直接调用 `sort()` 而不传递任何参数（即没有“比较函数”）时，JavaScript 引擎会执行以下步骤：

1. 将数组中的每一个元素都调用其 `toString()` 方法，转换为**字符串**。
    
2. 按照字符串在 **Unicode 字符表**中的顺序进行排列
```js
var threeSum = function(nums) {

    //双指针

    const res=[]

    nums.sort((a,b)=>{return a-b})

    for(let i=0;i<nums.length;i++){

        if(i>0&&nums[i]===nums[i-1]) continue//找到不重复的第一个元素

        let l=i+1,r=nums.length-1//右侧双指针

        while(l<r){

            if(nums[l]+nums[r]===-nums[i]){

                res.push([nums[i],nums[l],nums[r]])
				//循环结束后 `l` 指向的是最后一个重复元素
				//相等时 不会再有下一种组合了 我们左右指针都要修改
                while(nums[l]===nums[l+1]&&l<r) l++

                while(nums[r]===nums[r-1]&&l<r) r--

                l++

                r--

            }

            else{

                if(nums[l]+nums[r]<-nums[i]){

                    l++

                }

                else{

                    r--

                }

            }

        }

  

    }

    return res

};
```
