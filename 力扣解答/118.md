var generate = function(numRows) {

    const ans=[]

    ans.push([1])

    for(let k=1;k<numRows;k++){

        let cur=new Array(k+1).fill(1)

        for (let index = 1; index < cur.length-1; index++) {
			//重点是杨辉三角 这一行这里
            cur[index]=ans[k-1][index-1]+ans[k-1][index]

        }

        ans.push(cur)

    }

    return ans

};
