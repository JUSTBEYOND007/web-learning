和完全背包不同

完全背包是同一个物品连续选择，然后就再也不选这个物品了。本题可以交替选。比如 s 是 ABA 型，如果用完全背包的写法，只能枚举 AAB、ABB 这类连续的字符串组合，无法枚举到 ABA 这样的字符串组合。
外面 我们需要从前往后遍历 s  里面 我们是从后往前遍历 截取 配合前面的dp 来判断当前是否可以

思路：dp[i]表示 s[0..i-1] 这个前缀字符串是否可以被 wordDict 拆分

```js

var wordBreak = function(s, wordDict) {

    const dp=new Array(s.length+1).fill(false)

    let maxLen=Math.max(...wordDict.map(word=>word.length))

    dp[0]=true

    const set=new Set(wordDict)//用set来保存wordDict 用has判断节省时间复杂度 

    for(let i=1;i<=s.length;i++){//i表示的是字符串下标+1 i=0表示空时  

        for(let j=i-1;j>=Math.max(i-maxLen,0);j--){//我们从后往前切割 判断子字符串是否能被wordDict 拆分  j表示截取的初始位置
		//j>=Math.max(i-maxLen,0)表示 j的最小位置 j是指分割位置
            if(dp[j]&&set.has(s.slice(j,i))){
		//s.slice(j,i)取出字符串 s 中从 j 到 i−1 的子串 
                dp[i]=true

                break

            }

        }

    }

    return dp[s.length]

};
```
