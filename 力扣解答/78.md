<font color="#9B59B6">全部子集就是所有元素选上或者不选的所有组合</font>

- 每个元素都只有两种选择：  
    **选它 / 不选它**
    
- 所有选择组合的数量为 `2^n`

```js
var subsets = function(nums) {

    //for 遍历+回溯

    const result=[]

    const path=[]

  
//终止条件 走到最后了 cur===nums.length 所以我们需要cur来记录当前递归索引
    function backfall(cur){

        if(cur===nums.length)

            return  result.push([...path])

        path.push(nums[cur])//当前元素有两种选择 加入或者不加入

        backfall(cur+1)//只处理该层即可

        path.pop()

        backfall(cur+1)

    }

    backfall(0)

    return result

  

};
```
