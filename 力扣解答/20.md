我们在数组arr中只存左半边符号  如果最后数组长度为0则说明true
用map来保存左右符号匹配关系
```js

var isValid = function(s) {

    const arr=[]

    const map=new Map()

    map.set('(',')')

    map.set('[',']')

    map.set('{','}')

  

    for (let index = 0; index < s.length; index++) {

        const element = s.charAt(index);

        if(element==='('||element=='['||element=='{')

            arr.push(element)

        else {

            const char=arr.pop()

            if(map.get(char)!==element)

                return false

        }

    }

    return arr.length===0

};

```
