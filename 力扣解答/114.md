遍历找到左子树的右下角元素 然后我们将右下角元素的right指向根节点的右子树 把左子树挂在根节点的右子树上 把根节点的左子树值为null    即把它的左子树“插”到右子树前面   然后我们将cur指向右子树上（即原来的左子树）

并不是先序排列的解法 而是对应的是 寻找前驱节点 方法
每个元素遍历一遍O(n) 空间复杂度O(1)
```
var flatten = function(root) {

    let cur=root//用cur来遍历

    while(cur){

        if(cur.left){

            let p=cur.left

            while(p.right){

                p=p.right//找到左子树最右下角的元素

            }

        p.right=cur.right//把左子树“插”到右子树前面

        cur.right=cur.left//整体挂在右子树上

        cur.left=null//左子树置为null

        }

        cur=cur.right//遍历

    }

  

};
```
