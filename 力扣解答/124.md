class Solution {

     int res=Integer.MIN_VALUE;//res来存最大路径和 因为要递归 我们直接定义为成员变量

    public int maxPathSum(TreeNode root) {

        res=root.val;

        dfs(root);//调用dfs  不需要用值来接受 并不是存最大路径和

        return res;

    }

  

    public int dfs(TreeNode root){

        if(root==null)

            return 0;//注意  都是先判断是否return

        int lmax=Math.max(0,dfs(root.left));//判断左枝和0比较 因为可能为负数 很巧!

        int rmax=Math.max(0,dfs(root.right));

        res=Math.max(res,root.val+lmax+rmax);//判断以此为树根节点的最大值

        return root.val+Math.max(lmax,rmax);
	//注意这里是root.val+Math.max(lmax,rmax)而不是Math.max(res,root.val+lmax+rmax)  因为我们这里需要返回的是作为“从当前节点向上延申能组成的最大单边路径和”， 而不是“当前节点为中心的最大路径”  return的并不是res 我们要的最终值！！！
    }

}
