回溯就是递归遍历所有的情况 是否是回文串是我们是否继续递归的判断条件  是剪枝条件  不需要联合起来把问题复杂化

```js
//回溯
var partition = function(s) {

    const res=[]

    const path=[]

    traceback(0)

    return res
	//index表示分割起始位置 
    function traceback(index){

        if(index>=s.length){//说明分割完了 且字串如何回文串 加入res中

            res.push([...path])

            return

        }

        for (let i = index; i < s.length; i++) {

            if(isHW(index,i)){//从i = index开始分割  i表示分割终点

                path.push(s.substring(index,i+1))

                traceback(i+1)

                path.pop()

            }

        }

    }

    function isHW(left,right){

        while(left<=right){

            if(s.charAt(left)===s.charAt(right)){

                left++

                right--

            }else return false

        }

        return true

    }

  

};
```


优化版本 
我们每次有重复的判断 执行isHW(left,right)  left,right字串是否为回文串 我们可以用dp来记录 
用空间来换减少时间

```js
var partition = function(s) {

    const res=[]

    const path=[]

    const dp=Array.from({length:s.length},()=>new Array(s.length).fill(false))

    traceback(0)

    return res

    function traceback(index){

        if(index>=s.length){

            res.push([...path])

            return

        }

        for (let i = index; i < s.length; i++) {

            if(isHW(index,i)){

                dp[index][i]=true

                path.push(s.substring(index,i+1))

                traceback(i+1)

                path.pop()

            }

        }

    }

    function isHW(left,right){

        if(dp[left][right]) return true

        let l=left,r=right

        while(left<=right){

            if(s.charAt(left)===s.charAt(right)){

                left++

                right--

            }else return false

        }

        dp[l][r]=true

        return true

    }
};
```
