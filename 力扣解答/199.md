```js
var rightSideView = function(root) {

    //bfs

    if(!root)  

        return []

    const res=[]

    const queue=[]

    queue.push(root)

  

    while(queue.length>0){

        let curLength=queue.length

        for(let i=0;i<curLength;i++){

            let cur=queue.shift()

            if(cur.left) queue.push(cur.left)

            if(cur.right) queue.push(cur.right)

            if(i===curLength-1){

                res.push(cur.val)

            }

        }

    }

    return res

};
```
