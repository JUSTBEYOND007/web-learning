重点是能理解到为是一个二叉搜索树  向左即变小  向下即变大
以右上角开始比较 进行判断
```js
var searchMatrix = function(matrix, target) {

    //从右上角(左下角看也是)看是一颗二叉搜索树

    //横向和纵向最多分别比较m 和 n次  时间复杂度为O(m+n)

    let rows=matrix.length-1,cols=matrix[0].length-1

    let startRow=0,startCol=cols

    while(startRow<=rows&&startCol>=0){

        if(matrix[startRow][startCol]===target)

            return true

        else if(matrix[startRow][startCol]<target){

            startRow++

        }else{

            startCol--

        }

    }

    return false

};
```
