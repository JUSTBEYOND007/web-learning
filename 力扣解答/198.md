偷前k-1间房子 是指从前面k-1间房子中偷 

```js
var rob = function(nums) {

    const dp=new Array(nums.length)

    dp[0]=nums[0]

    if(nums.length===1)

        return dp[0] //因为我们需要用到i-2

    dp[1]=Math.max(nums[0],nums[1])

    for(let i=2;i<nums.length;i++){

        dp[i]=Math.max(dp[i-1],dp[i-2]+nums[i])

    }

    return dp[nums.length-1]

};
```

我们这里计算dp[i]只需要知道dp[i-1],dp[i-2] 即前两个 状态
我们可以用变量来保存即可 向后遍历时我们更新变量 即可 这样时间复杂度为O(1)
```js
//这里f0和f1分别表示dp[i-2] 和dp[i-1]
var rob = function(nums) {

    let f0 = nums[0];

    if (nums.length === 1)

        return f0; // 因为我们需要用到 i-2

  

    let f1 = Math.max(nums[0], nums[1]);

    let cur_f=f1//初始化为f1 解决了 nums.length=2的问题

    for (let i = 2; i < nums.length; i++) {

        cur_f = Math.max(f1, f0 + nums[i]);

        f0=f1

        f1=cur_f

    }

  

    return cur_f;

};
```
