26-7-23 不太会的题
严格使用中序遍历   重点时要记得用闭包pre来记录前面的那个值
思路：中序遍历 如果得到的树的数组是单调递增的  说明是二叉搜索树
我们用pre来保存前面那个值 来进行比较

```
        var isValidBST = function(root) {

            let pre=-Number.MAX_VALUE//使用了闭包 Infinity表示无穷大

            function dfs(root){

                if(root==null)

                    return true

                if(!dfs(root.left))

                    return false

                if(root.val<=pre)//暴增严格递增

                    return false

                pre=root.val //更新pre值

                return dfs(root.right)//就只剩这种情况 直接return即可

            }

            return  dfs(root)

        };
```
