题目“请你判断是否可以将这个数组分割成两个子集，使得两个子集的元素和相等。”
就是是否我们选择数组中的部分元素 求和为总和的一半 就是01背包问题

```js
var canPartition = function(nums) {

    let sum=nums.reduce((pre,value)=>pre=pre+value,0)//reduce使用

    if(sum%2!==0)

        return false

    let target=sum/2

    const dp=new Array(target+1).fill(false)

    dp[0]=true

    for (const element of nums) {

        for(let j=target;j>=element;j--){//注意j>=element否则dp[j-element]会越界 报错

            dp[j]=dp[j]||dp[j-element]

        }

    }

    return dp[target]

};
```


```java

class Solution {

    public boolean canPartition(int[] nums) {

        int total_sum=0;

        for (int num : nums) {

            total_sum+=num;

        }

        if(total_sum%2!=0)

            return false;

        int target=total_sum/2;//找到一个子集的和为总和的一半即可

        boolean dp[]=new boolean[target+1];//dp表示的是数组下标在nums数组总能否获得到  能得到为true

        dp[0]=true;

        for (int i = 0; i < nums.length; i++) {

            boolean dp_tmp[]=new boolean[target+1];

            System.arraycopy(dp,0,dp_tmp,0,dp.length);//复制数组 因为下面要对dp操作 整体上是一层一层遍历的

            for (int j = 0; j <= target; j++) {

                if(dp[j]&&j+nums[i]<=target)

                    dp_tmp[j+nums[i]]=true;

            }

            System.arraycopy(dp_tmp,0,dp,0,dp.length);

            if(dp[target]==true)

                return  true;

        }

        return  dp[target];

    }

}
```
