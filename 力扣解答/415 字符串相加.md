我们使用双指针来从后往前指


```js
var addStrings = function(num1, num2) {

    let i=num1.length-1

    let j=num2.length-1

    let carry=0

    let res=''

    while(i>=0||j>=0||carry){//注意条件i>=0 而不是i==0 i可以为0  同时注意还有carry也是条件 当carry不为0时  还需要进位1

        let cur_1=i>=0? Number(num1.charAt(i)):0

        let cur_2=j>=0? Number(num2.charAt(j)):0

        let num=(cur_1+cur_2+carry)%10//当前位置时%号

        carry=Math.floor((cur_1+cur_2+carry)/10)

        res=num+res

        i--

        j--

    }

    return res

};
```
