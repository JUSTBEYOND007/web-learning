所以 这里本题74是240的子集  所以可以直接使用240的解法

  var searchMatrix = function(matrix, target) {

    while(startRow<=rows&&startCol>=0){//只会遍历一行一列

        else if(matrix[startRow][startCol]<target){

            startRow++

    }

};

或者本题使用二分查找 先判断在某一行 再在该行判断  代码复杂
我们可以将每一行的数据拼接为一行 这样也保持着升序 我们使用一次二分查找即可
我们使用  let curRow=Math.floor(mid/cols),curCol=mid%cols 来判断行数和列数
注意是使用cols （列数 即每行个数 来判断当前行和列） Math.floor取值floor因为是从0开始的
```js
var searchMatrix = function(matrix, target) {

    const rows=matrix.length,cols=matrix[0].length

    let left=0,right=rows*cols-1

    while(left<=right){

        let mid=Math.floor((left+right)/2)

        let curRow=Math.floor(mid/cols),curCol=mid%cols

        if(matrix[curRow][curCol]===target){

            return true

        }else if(matrix[curRow][curCol]<target)

            left=mid+1

        else right=mid-1

    }

    return false

};
```
