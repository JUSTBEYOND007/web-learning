思路还是挺简单的 我们要计算time  正好和广度优先遍历 遍历每一层时间+1
思路就是bfs  记得要记录新鲜橘子个数  因为可能存在最后新鲜橘子不会被腐烂返回-1的情况
同时注意循环条件是   while(queue.length>0&&fresh>0) 表示只要还有新鲜橘子没被感染，才继续“过一分钟”
否则最后都腐烂了 我们还会while循环一次 time不对 
或者我们用flag 来记录 只有有新鲜橘子腐烂了 我们才time++


自己老是while死循环 注意逻辑 腐烂要让新鲜橘子变为2 否则会一直 被“相邻的腐烂橘子”重复感染
导致 **反复入队**
```js
var orangesRotting = function(grid) {

    //bfs

    //先记录新鲜橘子和腐烂橘子加入到队列中

    if(!grid||grid.length===0)

        return 0

    const queue=[]

    let time=0

    let fresh=0

    for(let i=0;i<grid.length;i++){

        for(let j=0;j<grid[0].length;j++){

            if(grid[i][j]==2)

                queue.push([i,j])

            else if(grid[i][j]===1)

                fresh++

        }

    }

    const directions=[[0,1],[0,-1],[1,0],[-1,0]]

    while(queue.length>0&&fresh>0){

        let curLength=queue.length

        for(let i=0;i<curLength;i++){

            const [row,col]=queue.shift()

            for(const [dx,dy] of directions){

                let x=dx+row

                let y=dy+col

                if(x<0||x>=grid.length||y<0||y>=grid[0].length||grid[x][y]!==1)

                    continue

                grid[x][y]=2

                queue.push([x,y])

                fresh--

            }

        }

        time++

    }

    if(fresh!==0)

        return -1

    return time

};
```
