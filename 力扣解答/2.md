我们使用while (l1 || l2 || carry)  用if判断 代码简单很多
我们使用`dummyNode` 来使得不用关心「是不是第一个节点



`dummyNode` 的本质是什么？

> **一个“假头节点”，专门用来统一处理头节点的特殊情况**

它让你：

- 不用关心「是不是第一个节点」
    
- 永远写 `cur.next = new ListNode(...)`
```
var addTwoNumbers = function(l1, l2) {
  let carry = 0
  const dummyNode = new ListNode(0)
  let cur = dummyNode

  while (l1 || l2 || carry) {
    let sum = carry

    if (l1) {
      sum += l1.val
      l1 = l1.next
    }

    if (l2) {
      sum += l2.val
      l2 = l2.next
    }

    carry = Math.floor(sum / 10)
    cur.next = new ListNode(sum % 10)
    cur = cur.next
  }

  return dummyNode.next
}
```
