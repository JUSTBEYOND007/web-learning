首推‘’‘## 拼接 + 拆分’方法 空间复杂度为O(1)

```
var copyRandomList = function(head) {

    //第一次遍历 拼接对应的newNode

    let cur=head

    if(head==null) return null

    while(cur){

        const newNode=new ListNode(cur.val)

        let tmp=cur.next

        cur.next=newNode

        newNode.next=tmp

        cur=tmp

    }

    //第二次遍历 修改random值

    cur=head

    while(cur){

        let newNode=cur.next

        newNode.random=cur.random?cur.random.next:null

        cur=cur.next.next

    }

    //第三次遍历 修改next值 拆分新旧链表

    let oldCur = head

    let newHead = head.next

    let newCur = newHead

    while (oldCur) {
	//逻辑关系重要  这样不会断链的 
        oldCur.next = oldCur.next.next// oldCur.next比不可能为null 
        newCur.next = newCur.next ? newCur.next.next : null
        oldCur = oldCur.next
        newCur = newCur.next
    }


    return newHead

};
```
两次遍历 空间复杂度为O(n)
第一次我们创建map（原因：否则random指向的新节点 我们不知道如何连接起来）来保存对应当前节点和拷贝的新节点new ListNode(cur.val)
第二次 我们设置新节点链表的next和random值 （可以通过原链表的关系和map来获取）
```
var copyRandomList = function(head) {

    const map=new Map()

    let cur=head

    while(cur){

        map.set(cur,new ListNode(cur.val))//可以只赋值一个val?

        cur=cur.next

    }

    cur=head

    while(cur){

        const newNode=map.get(cur)

        newNode.next=cur.next?map.get(cur.next):null//防止是最后一个节点

        newNode.random=cur.random?map.get(cur.random):null

        cur=cur.next

    }

    return map.get(head)

};
```
