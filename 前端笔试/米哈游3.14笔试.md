```js
var lengthOfLongestSubstring = function(s) {

    //滑动窗口+map判断

    const map=new Map()

    let start=0,end=0

    let maxLength=0

    for(let i=0;i<s.length;i++){

        if(map.has(s.charAt(i))&&map.get(s.charAt(i))>=start){

            start=map.get(s.charAt(i))+1

            map.set(s.charAt(i),i)

        }else {

            map.set(s.charAt(i),i)

            maxLength=Math.max(maxLength,i-start+1)

        }

    }

    return maxLength

};
```

 1️⃣ 子集和问题（Subset Sum）
目标是：
> 判断是否能选出一些数，使得 **和等于某个值**（或 ≥ 某个值）。
特点：
- 每个元素只能选一次
- **关注的是“能不能组成某个和”**

```js
function minSumAtLeastT(nums, T) {
    const maxSum = nums.reduce((a,b)=>a+b,0)

    const dp = new Array(maxSum + 1).fill(false)
    dp[0] = true

    for (let num of nums) {
        for (let j = maxSum; j >= num; j--) {
            dp[j] = dp[j] || dp[j - num]
        }
    }

    for (let i = T; i <= maxSum; i++) {
        if (dp[i]) return i
    }

    return -1
}
```
